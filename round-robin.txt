#include <iostream>
using namespace std;

int main() {
    int n, quantum;
    cout << "Enter number of processes: ";
    cin >> n;
    cout << "Enter time quantum: ";
    cin >> quantum;

    int bt[n], rem_bt[n], wt[n], tat[n];
    for (int i = 0; i < n; i++) {
        cout << "Burst time P" << i + 1 << ": ";
        cin >> bt[i];
        rem_bt[i] = bt[i];
    }

    int t = 0;
    while (true) {
        bool done = true;
        for (int i = 0; i < n; i++) {
            if (rem_bt[i] > 0) {
                done = false;
                if (rem_bt[i] > quantum) {
                    t += quantum;
                    rem_bt[i] -= quantum;
                } else {
                    t += rem_bt[i];
                    wt[i] = t - bt[i];
                    rem_bt[i] = 0;
                }
            }
        }
        if (done) break;
    }

    for (int i = 0; i < n; i++) tat[i] = bt[i] + wt[i];
    cout << "P\tBT\tWT\tTAT\n";
    for (int i = 0; i < n; i++)
        cout << i + 1 << "\t" << bt[i] << "\t" << wt[i] << "\t" << tat[i] << "\n";
    return 0;
}
