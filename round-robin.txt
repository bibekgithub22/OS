#include<iostream>
using namespace std;

int main() {

    int n, tq;

    cout<<"Enter number of processes: ";
    cin>>n;

    int at[n], bt[n], rt[n];
    int ct[n], tat[n], wt[n];

    for(int i=0;i<n;i++) {

        cout<<"\nEnter AT and BT for P"<<i+1<<": ";
        cin>>at[i]>>bt[i];

        rt[i]=bt[i];
    }

    cout<<"\nEnter Time Quantum: ";
    cin>>tq;

    int time=0, completed=0;

    while(completed<n) {

        bool found=false;

        for(int i=0;i<n;i++) {

            if(at[i]<=time && rt[i]>0) {

                found=true;

                if(rt[i]>tq) {

                    time += tq;

                    rt[i] -= tq;
                }

                else {

                    time += rt[i];

                    ct[i]=time;

                    tat[i]=ct[i]-at[i];

                    wt[i]=tat[i]-bt[i];

                    rt[i]=0;

                    completed++;
                }
            }
        }

        if(found==false)
            time++;
    }

    float avgWT=0, avgTAT=0;

    cout<<"\nP\tAT\tBT\tCT\tTAT\tWT\n";

    for(int i=0;i<n;i++) {

        avgWT += wt[i];

        avgTAT += tat[i];

        cout<<"P"<<i+1<<"\t"
            <<at[i]<<"\t"
            <<bt[i]<<"\t"
            <<ct[i]<<"\t"
            <<tat[i]<<"\t"
            <<wt[i]<<endl;
    }

    cout<<"\nAverage WT = "<<avgWT/n;

    cout<<"\nAverage TAT = "<<avgTAT/n;
}
