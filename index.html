First-Come, First-Served (FCFS)

Processes are executed strictly in the order they arrive in the ready queue. Simple, non-preemptive, but can suffer from the convoy effect.

#include <iostream>
using namespace std;

int main() {
    int n;
    cout << "Enter number of processes: ";
    cin >> n;

    int bt[n], wt[n], tat[n];
    for (int i = 0; i < n; i++) {
        cout << "Burst time P" << i + 1 << ": ";
        cin >> bt[i];
    }

    wt[0] = 0;
    for (int i = 1; i < n; i++)
        wt[i] = wt[i - 1] + bt[i - 1];

    for (int i = 0; i < n; i++)
        tat[i] = wt[i] + bt[i];

    cout << "P\tBT\tWT\tTAT\n";
    for (int i = 0; i < n; i++)
        cout << i + 1 << "\t" << bt[i] << "\t" << wt[i] << "\t" << tat[i] << "\n";
    return 0;
}

Shortest Job First (SJF)

Non-preemptive scheduling that picks the process with the smallest burst time next. Optimal average waiting time when burst times are known.

#include <iostream>
#include <algorithm>
using namespace std;

int main() {
    int n;
    cout << "Enter number of processes: ";
    cin >> n;

    int bt[n], wt[n], tat[n], p[n];
    for (int i = 0; i < n; i++) {
        cout << "Burst time P" << i + 1 << ": ";
        cin >> bt[i];
        p[i] = i + 1;
    }

    for (int i = 0; i < n - 1; i++)
        for (int j = 0; j < n - i - 1; j++)
            if (bt[j] > bt[j + 1]) {
                swap(bt[j], bt[j + 1]);
                swap(p[j], p[j + 1]);
            }

    wt[0] = 0;
    for (int i = 1; i < n; i++)
        wt[i] = wt[i - 1] + bt[i - 1];

    for (int i = 0; i < n; i++)
        tat[i] = wt[i] + bt[i];

    cout << "P\tBT\tWT\tTAT\n";
    for (int i = 0; i < n; i++)
        cout << p[i] << "\t" << bt[i] << "\t" << wt[i] << "\t" << tat[i] << "\n";
    return 0;
}

Shortest Remaining Time First (SRTF)

Preemptive variant of SJF — at every time unit, the process with the smallest remaining burst time is selected.

#include <iostream>
#include <climits>
using namespace std;

int main() {
    int n;
    cout << "Enter number of processes: ";
    cin >> n;

    int bt[n], at[n], rt[n], wt[n], tat[n];
    for (int i = 0; i < n; i++) {
        cout << "Arrival, Burst P" << i + 1 << ": ";
        cin >> at[i] >> bt[i];
        rt[i] = bt[i];
    }

    int complete = 0, t = 0, minm = INT_MAX, shortest = 0, finish_time;
    bool check = false;

    while (complete != n) {
        for (int j = 0; j < n; j++) {
            if (at[j] <= t && rt[j] < minm && rt[j] > 0) {
                minm = rt[j];
                shortest = j;
                check = true;
            }
        }
        if (!check) { t++; continue; }
        rt[shortest]--;
        minm = rt[shortest] ? rt[shortest] : INT_MAX;
        if (rt[shortest] == 0) {
            complete++;
            check = false;
            finish_time = t + 1;
            wt[shortest] = finish_time - bt[shortest] - at[shortest];
            if (wt[shortest] < 0) wt[shortest] = 0;
        }
        t++;
    }

    for (int i = 0; i < n; i++) tat[i] = bt[i] + wt[i];
    cout << "P\tBT\tWT\tTAT\n";
    for (int i = 0; i < n; i++)
        cout << i + 1 << "\t" << bt[i] << "\t" << wt[i] << "\t" << tat[i] << "\n";
    return 0;
}

Round Robin (RR)

Each process gets a fixed time slice (quantum) in cyclic order. Fair, preemptive, and widely used in time-sharing systems.

#include <iostream>
using namespace std;

int main() {
    int n, quantum;
    cout << "Enter number of processes: ";
    cin >> n;
    cout << "Enter time quantum: ";
    cin >> quantum;

    int bt[n], rem_bt[n], wt[n], tat[n];
    for (int i = 0; i < n; i++) {
        cout << "Burst time P" << i + 1 << ": ";
        cin >> bt[i];
        rem_bt[i] = bt[i];
    }

    int t = 0;
    while (true) {
        bool done = true;
        for (int i = 0; i < n; i++) {
            if (rem_bt[i] > 0) {
                done = false;
                if (rem_bt[i] > quantum) {
                    t += quantum;
                    rem_bt[i] -= quantum;
                } else {
                    t += rem_bt[i];
                    wt[i] = t - bt[i];
                    rem_bt[i] = 0;
                }
            }
        }
        if (done) break;
    }

    for (int i = 0; i < n; i++) tat[i] = bt[i] + wt[i];
    cout << "P\tBT\tWT\tTAT\n";
    for (int i = 0; i < n; i++)
        cout << i + 1 << "\t" << bt[i] << "\t" << wt[i] << "\t" << tat[i] << "\n";
    return 0;
}
