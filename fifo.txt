#include<iostream>
using namespace std;

int main() {

    int n, frames;

    cout<<"Enter number of pages: ";
    cin>>n;

    int pages[n];

    cout<<"Enter page sequence: ";

    for(int i=0;i<n;i++) {
        cin>>pages[i];
    }

    cout<<"Enter number of frames: ";
    cin>>frames;

    int f[frames];

    for(int i=0;i<frames;i++)
        f[i]=-1;

    int index=0, faults=0;

    for(int i=0;i<n;i++) {

        bool found=false;

        for(int j=0;j<frames;j++) {

            if(f[j]==pages[i]) {

                found=true;
                break;
            }
        }

        if(!found) {

            f[index]=pages[i];

            index=(index+1)%frames;

            faults++;
        }
    }

    cout<<"\nPage Faults = "<<faults;

    return 0;
}
