#include <iostream>

using namespace std;

int main() {
    int n, r, i, j;
    cout << "Enter number of processes: ";
    cin >> n;
    cout << "Enter number of resources: ";
    cin >> r;

    // Dynamic allocation using new
    int** alloc = new int*[n];
    int** maxNeed = new int*[n];
    int** need = new int*[n];
    
    for (i = 0; i < n; i++) {
        alloc[i] = new int[r];
        maxNeed[i] = new int[r];
        need[i] = new int[r];
    }

    int* available = new int[r];
    int* work = new int[r];
    int* finish = new int[n];
    int* safeSeq = new int[n];

    cout << "Enter Allocation matrix:\n";
    for (i = 0; i < n; i++)
        for (j = 0; j < r; j++)
            cin >> alloc[i][j];

    cout << "Enter Max matrix:\n";
    for (i = 0; i < n; i++)
        for (j = 0; j < r; j++)
            cin >> maxNeed[i][j];

    cout << "Enter Available resources:\n";
    for (j = 0; j < r; j++)
        cin >> available[j];

    // Calculate need matrix
    for (i = 0; i < n; i++)
        for (j = 0; j < r; j++)
            need[i][j] = maxNeed[i][j] - alloc[i][j];

    // Display need matrix
    cout << "\nNeed Matrix:\n";
    for (i = 0; i < n; i++) {
        for (j = 0; j < r; j++)
            cout << need[i][j] << " ";
        cout << "\n";
    }

    // Initialize work and finish arrays
    for (j = 0; j < r; j++)
        work[j] = available[j];

    for (i = 0; i < n; i++)
        finish[i] = 0;

    // Banker's algorithm
    int count = 0;
    while (count < n) {
        int found = 0;
        for (i = 0; i < n; i++) {
            if (finish[i] == 0) {
                int ok = 1;
                for (j = 0; j < r; j++)
                    if (need[i][j] > work[j]) {
                        ok = 0;
                        break;
                    }
                if (ok) {
                    for (j = 0; j < r; j++)
                        work[j] += alloc[i][j];
                    finish[i] = 1;
                    safeSeq[count++] = i;
                    found = 1;
                }
            }
        }
        if (found == 0)
            break;
    }

    if (count == n) {
        cout << "\nSystem is in SAFE STATE\n";
        cout << "Safe Sequence: ";
        for (i = 0; i < n; i++)
            cout << "P" << safeSeq[i] << " ";
        cout << "\n";
    } else {
        cout << "\nSystem is in UNSAFE STATE\n";
        cout << "Deadlock may occur!\n";
    }

    // Deallocate memory
    for (i = 0; i < n; i++) {
        delete[] alloc[i];
        delete[] maxNeed[i];
        delete[] need[i];
    }
    delete[] alloc;
    delete[] maxNeed;
    delete[] need;
    delete[] available;
    delete[] work;
    delete[] finish;
    delete[] safeSeq;

    return 0;
}
