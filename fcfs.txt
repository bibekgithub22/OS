#include<iostream>
using namespace std;

int main() {

    int n;

    cout<<"Enter number of processes: ";
    cin>>n;

    int at[n], bt[n], ct[n], tat[n], wt[n];

    // Input
    for(int i=0;i<n;i++) {

        cout<<"\nEnter AT and BT for P"<<i+1<<": ";
        cin>>at[i]>>bt[i];
    }

    // Completion Time
    ct[0]=at[0]+bt[0];

    for(int i=1;i<n;i++) {

        if(ct[i-1]<at[i])
            ct[i]=at[i]+bt[i];
        else
            ct[i]=ct[i-1]+bt[i];
    }

    float avgWT=0, avgTAT=0;

    // TAT and WT
    for(int i=0;i<n;i++) {

        tat[i]=ct[i]-at[i];

        wt[i]=tat[i]-bt[i];

        avgWT += wt[i];

        avgTAT += tat[i];
    }

    // Output
    cout<<"\nP\tAT\tBT\tCT\tTAT\tWT\n";

    for(int i=0;i<n;i++) {

        cout<<"P"<<i+1<<"\t"
            <<at[i]<<"\t"
            <<bt[i]<<"\t"
            <<ct[i]<<"\t"
            <<tat[i]<<"\t"
            <<wt[i]<<endl;
    }

    cout<<"\nAverage WT = "<<avgWT/n;

    cout<<"\nAverage TAT = "<<avgTAT/n;
}
